cmake_minimum_required(VERSION 3.20)
project(neonethack VERSION 1.0.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(EMSCRIPTEN)
  include(wasm/CMakeLists.txt)
  return()
endif()
set(LUA_HOME "$ENV{LUA_HOME}" CACHE PATH "Lua 5.4 development installation")
option(NNH_INSTALL_ENGINE "Install the built engine and static data with the library" ON)
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
include(CMakePackageConfigHelpers)
# Compile once; private regression tests link these objects directly instead
# of requiring private symbols to leak from the installed shared library.
add_library(neonethack-objects OBJECT src/api.c src/schema.c src/affordance.c src/explorer.c src/minjson.c src/session.c)
target_include_directories(neonethack-objects PRIVATE include src)
target_link_libraries(neonethack-objects PRIVATE Threads::Threads)
set_target_properties(neonethack-objects PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
add_library(neonethack $<TARGET_OBJECTS:neonethack-objects>)
target_link_libraries(neonethack PRIVATE Threads::Threads)
target_include_directories(neonethack PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(neonethack PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 2)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(neonethack-objects PRIVATE -Wall -Wextra "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
endif()
add_executable(neonethack-mcp cli/mcp.c cli/compact.c src/minjson.c)
target_include_directories(neonethack-mcp PRIVATE src)
target_link_libraries(neonethack-mcp PRIVATE neonethack)
add_executable(neonethack-cli cli/main.c)
target_link_libraries(neonethack-cli PRIVATE neonethack)
set_target_properties(neonethack-cli PROPERTIES OUTPUT_NAME neonethack)
# A versioned install prefix must work without LD_LIBRARY_PATH or ldconfig.
# Compute the relative path rather than assuming bin/ and lib/ defaults.
file(RELATIVE_PATH NNH_CLI_TO_LIB "/${CMAKE_INSTALL_BINDIR}" "/${CMAKE_INSTALL_LIBDIR}")
if(APPLE)
  set_property(TARGET neonethack-cli neonethack-mcp PROPERTY INSTALL_RPATH "@loader_path/${NNH_CLI_TO_LIB}")
elseif(UNIX)
  set_property(TARGET neonethack-cli neonethack-mcp PROPERTY INSTALL_RPATH "$ORIGIN/${NNH_CLI_TO_LIB}")
endif()
add_custom_target(engine
  COMMAND ${CMAKE_COMMAND} -E env "LUA_HOME=${LUA_HOME}" "CC=${CMAKE_C_COMPILER}" sh ${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-engine.sh
  COMMAND ${CMAKE_COMMAND} "-DROOT=${CMAKE_CURRENT_SOURCE_DIR}" "-DOUT=${CMAKE_BINARY_DIR}/data" -P ${CMAKE_CURRENT_SOURCE_DIR}/scripts/stage-data.cmake
  USES_TERMINAL VERBATIM)
add_custom_target(native DEPENDS neonethack-cli neonethack-mcp engine)
if(BUILD_TESTING)
  add_executable(neonethack-affordance-test tests/affordance.c src/affordance.c src/minjson.c)
  target_include_directories(neonethack-affordance-test PRIVATE src)
  target_compile_options(neonethack-affordance-test PRIVATE -UNDEBUG)
  add_test(NAME pure-affordance COMMAND neonethack-affordance-test)
  add_executable(neonethack-c-test tests/api.c)
  target_link_libraries(neonethack-c-test PRIVATE neonethack)
  add_test(NAME public-c-api COMMAND neonethack-c-test ${CMAKE_CURRENT_SOURCE_DIR}/engine/playground/nethack ${CMAKE_CURRENT_SOURCE_DIR}/engine/playground)
  add_executable(neonethack-core-test tests/core-regression.c $<TARGET_OBJECTS:neonethack-objects>)
  target_include_directories(neonethack-core-test PRIVATE src)
  target_link_libraries(neonethack-core-test PRIVATE Threads::Threads)
  add_test(NAME core-regression COMMAND neonethack-core-test ${CMAKE_CURRENT_SOURCE_DIR}/engine/playground/nethack ${CMAKE_CURRENT_SOURCE_DIR}/engine/playground)
  add_executable(neonethack-pipe-peer tests/pipe-peer.c)
  add_executable(neonethack-signal-test tests/session-signal.c $<TARGET_OBJECTS:neonethack-objects>)
  target_include_directories(neonethack-signal-test PRIVATE src)
  target_link_libraries(neonethack-signal-test PRIVATE Threads::Threads)
  add_test(NAME host-sigpipe COMMAND neonethack-signal-test $<TARGET_FILE:neonethack-pipe-peer>)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(neonethack-fault MODULE tests/storage-fault.c)
    target_link_libraries(neonethack-fault PRIVATE ${CMAKE_DL_LIBS})
    add_executable(neonethack-spawn-peer tests/spawn-peer.c)
    add_executable(neonethack-spawn-test tests/session-spawn.c src/session.c)
    target_include_directories(neonethack-spawn-test PRIVATE src)
    target_link_libraries(neonethack-spawn-test PRIVATE Threads::Threads)
    target_compile_options(neonethack-spawn-test PRIVATE -UNDEBUG)
    target_link_options(neonethack-spawn-test PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=strdup -Wl,--wrap=getenv -Wl,--wrap=setenv)
    add_test(NAME host-fork-and-stdio COMMAND neonethack-spawn-test $<TARGET_FILE:neonethack-spawn-peer>)
  endif()
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    foreach(target neonethack-c-test neonethack-core-test neonethack-signal-test)
      target_compile_options(${target} PRIVATE -UNDEBUG)
    endforeach()
  endif()
endif()
install(TARGETS neonethack neonethack-cli neonethack-mcp EXPORT neonethackTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/neonethack.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT neonethackTargets FILE neonethackTargets.cmake NAMESPACE neonethack:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/neonethack)
configure_package_config_file(scripts/neonethackConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/neonethackConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/neonethack)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/neonethackConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
configure_file(scripts/neonethack.pc.in ${CMAKE_CURRENT_BINARY_DIR}/neonethack.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/neonethackConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/neonethackConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/neonethack)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/neonethack.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
install(FILES protocol/mcp-response.schema.json protocol/catalog.json protocol/request.schema.json protocol/response.schema.json DESTINATION ${CMAKE_INSTALL_DATADIR}/neonethack/protocol)
install(FILES README.md NOTICE.md engine/dat/license DESTINATION ${CMAKE_INSTALL_DATADIR}/neonethack)
if(NNH_INSTALL_ENGINE)
  install(PROGRAMS engine/playground/nethack DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/neonethack RENAME engine)
  install(DIRECTORY ${CMAKE_BINARY_DIR}/data/ DESTINATION ${CMAKE_INSTALL_DATADIR}/neonethack/data)
  install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/build/LUA-LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/neonethack)
endif()
